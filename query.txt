-- CREATE gold_inspection_enquiries TABLE
CREATE TABLE `gold_inspection_enquiries` (
  `id` bigint(20) unsigned NOT NULL AUTO_INCREMENT,
  `name` varchar(255) NOT NULL,
  `phone` varchar(255) NOT NULL,
  `address` varchar(255) NOT NULL,
  `approx_grams` decimal(10,2) NOT NULL,
  `gold_type` varchar(255) NOT NULL,
  `gold_location` varchar(255) NOT NULL,
  `preferred_date` date NOT NULL,
  `photos` json DEFAULT NULL,
  `status` varchar(30) NOT NULL DEFAULT 'New',
  `admin_remark` text DEFAULT NULL,
  `reviewed_by` bigint(20) unsigned DEFAULT NULL,
  `reviewed_at` datetime DEFAULT NULL,
  `created_by_id` bigint(20) unsigned DEFAULT NULL,
  `updated_by_id` bigint(20) unsigned DEFAULT NULL,
  `created_at` timestamp NULL DEFAULT NULL,
  `updated_at` timestamp NULL DEFAULT NULL,
  `deleted_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `gold_inspection_enquiries_reviewed_by_foreign` (`reviewed_by`),
  KEY `gold_inspection_enquiries_created_by_id_foreign` (`created_by_id`),
  KEY `gold_inspection_enquiries_updated_by_id_foreign` (`updated_by_id`),
  CONSTRAINT `gold_inspection_enquiries_created_by_id_foreign` FOREIGN KEY (`created_by_id`) REFERENCES `users` (`id`) ON DELETE SET NULL,
  CONSTRAINT `gold_inspection_enquiries_reviewed_by_foreign` FOREIGN KEY (`reviewed_by`) REFERENCES `users` (`id`) ON DELETE SET NULL,
  CONSTRAINT `gold_inspection_enquiries_updated_by_id_foreign` FOREIGN KEY (`updated_by_id`) REFERENCES `users` (`id`) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- SEED MODULE & PERMISSIONS
INSERT INTO `modules` (`name`, `slug`, `status`, `created_at`, `updated_at`) 
VALUES ('Inspection', 'inspection', 'active', NOW(), NOW())
ON DUPLICATE KEY UPDATE `status`='active', `updated_at`=NOW();

-- Insert permissions
INSERT INTO `permissions` (`module_id`, `name`, `slug`, `created_at`, `updated_at`)

((SELECT `id` FROM `modules` WHERE `slug`='inspection'), 'View Inspection', 'inspection.view', NOW(), NOW()),
((SELECT `id` FROM `modules` WHERE `slug`='inspection'), 'Create Inspection', 'inspection.create', NOW(), NOW()),
((SELECT `id` FROM `modules` WHERE `slug`='inspection'), 'Edit Inspection', 'inspection.edit', NOW(), NOW()),
((SELECT `id` FROM `modules` WHERE `slug`='inspection'), 'Delete Inspection', 'inspection.delete', NOW(), NOW())
ON DUPLICATE KEY UPDATE `updated_at`=NOW();

-- ASSIGN TO SUPER ADMIN AND ADMIN ROLES
INSERT INTO `role_permissions` (`role_id`, `permission_id`, `allowed`)
SELECT r.id, p.id, 1 
FROM `roles` r
CROSS JOIN `permissions` p
WHERE r.slug IN ('super-admin', 'admin') 
AND p.slug IN ('inspection.view', 'inspection.create', 'inspection.edit', 'inspection.delete')
ON DUPLICATE KEY UPDATE `allowed`=1;
